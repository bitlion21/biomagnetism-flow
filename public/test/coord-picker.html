  <h2>Selector de coordenadas (1024×1024)</h2>

  <div class="row">

    <div class="panel">

      <img id="img" src="" alt="template" />

      <div class="hint">Haz clic sobre el punto exacto. Se guardará x,y en píxeles (0..1023).</div>

    </div>

    <div class="side">

      <label>Template (ruta relativa desde /public):</label>

      <input id="template" placeholder="/templates/master-femenino-con-organos.png" />

      

      <label>Nombre del punto:</label>

      <input id="point" placeholder="Apéndice" />

      

      <button id="load">Cargar template</button>

      <button id="copy">Copiar JSON al portapapeles</button>


      <label>JSON acumulado:</label>

      <textarea id="out" readonly></textarea>

      <div class="hint">Luego este JSON lo convierte Codex a XLSX o lo usa directamente.</div>

    </div>

  </div>


<script>

  const img = document.getElementById('img');

  const template = document.getElementById('template');

  const point = document.getElementById('point');

  const out = document.getElementById('out');

  const store = [];


  document.getElementById('load').onclick = () => {

    img.src = template.value.trim();

  };


  img.addEventListener('click', (e) => {

    if (!img.src) return;

    const rect = img.getBoundingClientRect();

    const x = Math.round((e.clientX - rect.left) * (1024 / rect.width));

    const y = Math.round((e.clientY - rect.top)  * (1024 / rect.height));

    const rec = { punto: point.value.trim(), template: template.value.trim(), x, y };

    store.push(rec);

    out.value = JSON.stringify(store, null, 2);

  });


  document.getElementById('copy').onclick = async () => {

    await navigator.clipboard.writeText(out.value);

    alert("JSON copiado.");

  };

</script>
